79inline StkFloat PitShift :: tick( StkFloat input )
80{
81 // Calculate the two delay length values, keeping them within the
82 // range 12 to maxDelay-12.
83 delay_[0] += rate_;
84 while ( delay_[0] > maxDelay-12 ) delay_[0] -= delayLength_;
85 while ( delay_[0] < 12 ) delay_[0] += delayLength_;
86
87 delay_[1] = delay_[0] + halfLength_;
88 while ( delay_[1] > maxDelay-12 ) delay_[1] -= delayLength_;
89 while ( delay_[1] < 12 ) delay_[1] += delayLength_;
90
91 // Set the new delay line lengths.
92 delayLine_[0].setDelay( delay_[0] );
93 delayLine_[1].setDelay( delay_[1] );
94
95 // Calculate a triangular envelope.
96 env_[1] = std::fabs( ( delay_[0] - halfLength_ + 12.0f ) * ( 1.0f / (halfLength_ + 12.0f ) ) );
97 env_[0] = 1.0f - env_[1];
98
99 // Delay input and apply envelope.
100 lastFrame_[0] = env_[0] * delayLine_[0].tick( input );
101 lastFrame_[0] += env_[1] * delayLine_[1].tick( input );
102
103 // Compute effect mix and output.
104 lastFrame_[0] *= effectMix_;
105 lastFrame_[0] += ( 1.0f - effectMix_ ) * input;
106
107 return lastFrame_[0];
108}
