194inline StkFrames& Granulate :: tick( StkFrames& frames, unsigned int channel )
195{
196 unsigned int nChannels = lastFrame_.channels();
197#if defined(_STK_DEBUG_)
198 if ( channel > frames.channels() - nChannels ) {
199 oStream_ << "Granulate::tick(): channel and StkFrames arguments are incompatible!";
200 handleError( StkError::FUNCTION_ARGUMENT );
201 }
202#endif
203
204 StkFloat *samples = &frames[channel];
205 unsigned int j, hop = frames.channels() - nChannels;
206 for ( unsigned int i=0; i<frames.frames(); i++, samples += hop ) {
207 *samples++ = tick();
208 for ( j=1; j<nChannels; j++ )
209 *samples++ = lastFrame_[j];
210 }
211
212 return frames;
213}
