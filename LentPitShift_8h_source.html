112inline void LentPitShift::process()
113{
114 StkFloat x_t; // input coefficient
115 StkFloat x_t_T; // previous input coefficient at T samples
116 StkFloat coeff; // new coefficient for the difference function
117
118 unsigned long alternativePitch = tMax_; // Global minimum storage
119 lastPeriod_ = tMax_+1; // Storage of the lowest local minimum under the threshold
120
121 // Loop variables
122 unsigned long delay_;
123 unsigned int n;
124
125 // Initialization of the dt coefficients. Since the
126 // frames are of tMax_ length, there is no overlapping
127 // between the successive windows where pitch tracking
128 // is performed.
129 for ( delay_=1; delay_<=tMax_; delay_++ )
130 dt[delay_] = 0.;
131
132 // Calculation of the dt coefficients and update of the input delay line.
133 for ( n=0; n<inputFrames.size(); n++ ) {
134 x_t = inputLine_.tick( inputFrames[ n ] );
135 for ( delay_=1; delay_<= tMax_; delay_++ ) {
136 x_t_T = inputLine_.tapOut( delay_ );
137 coeff = x_t - x_t_T;
138 dt[delay_] += coeff * coeff;
139 }
140 }
141
142 // Calculation of the pitch tracking function and test for the minima.
143 for ( delay_=1; delay_<=tMax_; delay_++ ) {
144 cumDt[delay_] = dt[delay_] + cumDt[delay_-1];
145 dpt[delay_] = dt[delay_] * delay_ / cumDt[delay_];
146
147 // Look for a minimum
148 if ( dpt[delay_-1]-dpt[delay_-2] < 0 && dpt[delay_]-dpt[delay_-1] > 0 ) {
149 // Check if the minimum is under the threshold
150 if ( dpt[delay_-1] < threshold_ ){
151 lastPeriod_ = delay_-1;
152 // If a minimum is found, we can stop the loop
153 break;
154 }
155 else if ( dpt[alternativePitch] > dpt[delay_-1] )
156 // Otherwise we store it if it is the current global minimum
157 alternativePitch = delay_-1;
158 }
159 }
160
161 // Test for the last period length.
162 if ( dpt[delay_]-dpt[delay_-1] < 0 ) {
163 if ( dpt[delay_] < threshold_ )
164 lastPeriod_ = delay_;
165 else if ( dpt[alternativePitch] > dpt[delay_] )
166 alternativePitch = delay_;
167 }
168
169 if ( lastPeriod_ == tMax_+1 )
170 // No period has been under the threshold so we used the global minimum
171 lastPeriod_ = alternativePitch;
172
173 // We put the new zero output coefficients in the output delay line and
174 // we get the previous calculated coefficients
175 outputLine_.tick( zeroFrame, outputFrames );
176
177 // Initialization of the Hamming window used in the algorithm
178 for ( int n=-(int)lastPeriod_; n<(int)lastPeriod_; n++ )
179 window[n+lastPeriod_] = (1 + std::cos(STK_PI*n/lastPeriod_)) / 2 ;
180
181 long M; // Index of reading in the input delay line
182 long N; // Index of writing in the output delay line
183 StkFloat sample; // Temporary storage for the new coefficient
184
185 // We loop for all the frames of length lastPeriod_ presents between inputPtr and tMax_
186 for ( ; inputPtr<(int)(tMax_-lastPeriod_); inputPtr+=lastPeriod_ ) {
187 // Test for the decision of compression/expansion
188 while ( outputPtr < inputPtr ) {
189 // Coefficients for the linear interpolation
190 env[1] = std::fmod( outputPtr + tMax_, 1.0 );
191 env[0] = 1.0f - env[1];
192 M = tMax_ - inputPtr + lastPeriod_ - 1; // New reading pointer
193 N = 2*tMax_ - (unsigned long)std::floor(outputPtr + tMax_) + lastPeriod_ - 1; // New writing pointer
194 for ( unsigned int j=0; j<2*lastPeriod_; j++,M--,N-- ) {
195 sample = inputLine_.tapOut(M) * window[j] / 2.f;
196 // Linear interpolation
197 outputLine_.addTo(env[0] * sample, N);
198 outputLine_.addTo(env[1] * sample, N-1);
199 }
200 outputPtr = outputPtr + lastPeriod_ * periodRatio_; // new output pointer
201 }
202 }
203 // Shifting of the pointers waiting for the new frame of length tMax_.
204 outputPtr -= tMax_;
205 inputPtr -= tMax_;
206}
