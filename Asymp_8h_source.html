102inline StkFloat Asymp :: tick( void )
103{
104 if ( state_ ) {
105
106 value_ = factor_ * value_ + constant_;
107
108 // Check threshold.
109 if ( target_ > value_ ) {
110 if ( target_ - value_ <= TARGET_THRESHOLD ) {
111 value_ = target_;
112 state_ = 0;
113 }
114 }
115 else {
116 if ( value_ - target_ <= TARGET_THRESHOLD ) {
117 value_ = target_;
118 state_ = 0;
119 }
120 }
121 lastFrame_[0] = value_;
122 }
123
124 return value_;
125}
