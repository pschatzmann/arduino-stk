112inline StkFloat SingWave :: tick( void )
113{
114 // Set the wave rate.
115 StkFloat newRate = pitchEnvelope_.tick();
116 newRate += newRate * modulator_.tick();
117 wave_.setRate( newRate );

119 lastFrame_[0] = wave_.tick();
