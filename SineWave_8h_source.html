99inline StkFloat SineWave :: tick( void )
100{
101 // Prevent that we crash if we call tick if it has not been set up yet
102 if (table_.size()==0) return 0.0f;
103
104 // Check limits of time address ... if necessary, recalculate modulo
105 // TABLE_SIZE.
106 while ( time_ < 0.0f )
107 time_ += TABLE_SIZE;
108 while ( time_ >= TABLE_SIZE )
109 time_ -= TABLE_SIZE;
110
111 iIndex_ = (unsigned int) time_;
112 alpha_ = time_ - iIndex_;
113 StkFloat tmp = table_[ iIndex_ ];
114 tmp += ( alpha_ * ( table_[ iIndex_ + 1 ] - tmp ) );
115
116 // Increment time, which can be negative.
117 time_ += rate_;
118
119 lastFrame_[0] = tmp;
120 return lastFrame_[0];
121}

132 StkFloat *samples = &frames[channel];
133 StkFloat tmp = 0.0;
134

136 for ( unsigned int i=0; i<frames.frames(); i++, samples += hop ) {
137

140 while ( time_ < 0.0f )
141 time_ += TABLE_SIZE;
142 while ( time_ >= TABLE_SIZE )
143 time_ -= TABLE_SIZE;

152 time_ += rate_;
